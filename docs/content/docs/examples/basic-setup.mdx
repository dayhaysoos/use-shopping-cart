A complete basic shopping cart implementation with add/remove functionality.

## Overview

This example demonstrates:

- Adding products to cart
- Removing items from cart
- Displaying cart contents
- Calculating totals
- Redirecting to checkout

## Complete Code

### App Setup

```jsx
// App.jsx
import { CartProvider } from 'use-shopping-cart'
import ProductList from './components/ProductList'
import Cart from './components/Cart'

function App() {
  return (
    <CartProvider
      stripe={process.env.REACT_APP_STRIPE_PUBLISHABLE_KEY}
      currency="USD"
    >
      <div className="app">
        <header>
          <h1>My Store</h1>
          <Cart />
        </header>
        <main>
          <ProductList />
        </main>
      </div>
    </CartProvider>
  )
}

export default App
```

### Product List Component

```jsx
// components/ProductList.jsx
import { useShoppingCart, formatCurrencyString } from 'use-shopping-cart'
import { products } from '../data/products'

function ProductList() {
  const { addItem } = useShoppingCart()

  return (
    <div className="product-list">
      <h2>Products</h2>
      <div className="products-grid">
        {products.map((product) => {
          const price = formatCurrencyString({
            value: product.price,
            currency: product.currency,
            language: 'en-US'
          })

          return (
            <div key={product.id} className="product-card">
              <img src={product.image} alt={product.name} />
              <h3>{product.name}</h3>
              <p>{product.description}</p>
              <p className="price">{price}</p>
              <button onClick={() => addItem(product)}>Add to Cart</button>
            </div>
          )
        })}
      </div>
    </div>
  )
}

export default ProductList
```

### Cart Component

```jsx
// components/Cart.jsx
import { useShoppingCart } from 'use-shopping-cart'

function Cart() {
  const {
    cartCount,
    formattedTotalPrice,
    cartDetails,
    incrementItem,
    decrementItem,
    removeItem,
    clearCart,
    redirectToCheckout
  } = useShoppingCart()

  if (cartCount === 0) {
    return (
      <div className="cart">
        <p>Your cart is empty</p>
      </div>
    )
  }

  return (
    <div className="cart">
      <h2>Shopping Cart ({cartCount} items)</h2>

      <div className="cart-items">
        {Object.entries(cartDetails).map(([id, item]) => (
          <div key={id} className="cart-item">
            <img src={item.image} alt={item.name} />
            <div className="item-details">
              <h4>{item.name}</h4>
              <p>{item.formattedPrice}</p>
            </div>

            <div className="quantity-controls">
              <button onClick={() => decrementItem(id)}>-</button>
              <span>{item.quantity}</span>
              <button onClick={() => incrementItem(id)}>+</button>
            </div>

            <p className="item-total">{item.formattedValue}</p>

            <button onClick={() => removeItem(id)}>Remove</button>
          </div>
        ))}
      </div>

      <div className="cart-footer">
        <div className="cart-total">
          <strong>Total:</strong>
          <strong>{formattedTotalPrice}</strong>
        </div>

        <div className="cart-actions">
          <button onClick={clearCart}>Clear Cart</button>
          <button onClick={() => redirectToCheckout()}>Checkout</button>
        </div>
      </div>
    </div>
  )
}

export default Cart
```

## Server Endpoint

Create a server endpoint to handle checkout:

```js
// pages/api/create-checkout-session.js (Next.js)
import Stripe from 'stripe'
const stripe = new Stripe(process.env.STRIPE_SECRET_KEY)

export default async function handler(req, res) {
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method not allowed' })
  }

  try {
    const { cartDetails } = req.body

    const line_items = Object.values(cartDetails).map((item) => ({
      price_data: {
        currency: 'usd',
        product_data: {
          name: item.name,
          images: [item.image]
        },
        unit_amount: item.price
      },
      quantity: item.quantity
    }))

    const session = await stripe.checkout.sessions.create({
      mode: 'payment',
      line_items,
      success_url: `${req.headers.origin}/success?session_id={CHECKOUT_SESSION_ID}`,
      cancel_url: req.headers.origin
    })

    res.redirect(303, session.url)
  } catch (error) {
    res.status(500).json({ error: error.message })
  }
}
```

## Key Concepts

### 1. Cart Persistence

The cart automatically persists to localStorage by default:

```jsx
<CartProvider
  stripe={STRIPE_KEY}
  currency="USD"
  shouldPersist={true} // Enable persistence (default)
  persistKey="my-cart" // localStorage key name (default: 'use-shopping-cart')
>
  <App />
</CartProvider>
```

### 2. Currency Formatting

Use `formatCurrencyString` for consistent formatting:

```jsx
import { formatCurrencyString } from 'use-shopping-cart'

const price = formatCurrencyString({
  value: 1000, // cents
  currency: 'USD',
  language: 'en-US'
})
// Returns: "$10.00"
```

### 3. Cart Details Structure

`cartDetails` is an object with product IDs as keys:

```js
{
  "banana_001": {
    id: "banana_001",
    name: "Bananas",
    price: 400,
    quantity: 2,
    value: 800, // price * quantity
    formattedPrice: "$4.00",
    formattedValue: "$8.00"
  }
}
```
