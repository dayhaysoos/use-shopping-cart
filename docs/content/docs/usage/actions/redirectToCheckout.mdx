Redirects the user to Stripe's Checkout page using a server-created session ID.

## Signature

```typescript
redirectToCheckout(sessionId: string): Promise<{ error: any } | undefined>
```

## Parameters

- `sessionId` (string, required): The checkout session ID returned from your server endpoint

## How It Works

1. Your server creates a Stripe checkout session
2. Server returns the `sessionId`
3. Call `redirectToCheckout(sessionId)` on the client
4. User is redirected to Stripe's checkout page

## Usage

```jsx
import React, { useState } from 'react'
import { useShoppingCart } from 'use-shopping-cart'
import { AddMoreItems } from './utilities'

export function RedirectToCheckout() {
  const [status, setStatus] = useState('idle')
  const { redirectToCheckout, cartCount } = useShoppingCart()

  async function handleClick(event) {
    event.preventDefault()

    if (cartCount > 0) {
      setStatus('loading')
      try {
        // 1. Call your server to create a checkout session
        const response = await fetch('/api/create-checkout-session', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            cartDetails: useShoppingCart().cartDetails
          })
        })

        const { sessionId } = await response.json()

        // 2. Redirect to Stripe checkout with the session ID
        const result = await redirectToCheckout(sessionId)
        if (result?.error) {
          console.error(result)
          setStatus('redirect-error')
        }
      } catch (error) {
        console.error(error)
        setStatus('redirect-error')
      }
    } else {
      setStatus('missing-items')
    }
  }

  return (
    <article
      style={{
        display: 'flex',
        flexDirection: 'column',
        justifyContent: 'center',
        alignItems: 'center',
        width: '50%'
      }}
    >
      {status === 'missing-items' && <AddMoreItems />}

      {status === 'redirect-error' && (
        <p>Unable to redirect to Stripe checkout page.</p>
      )}

      <button
        onClick={handleClick}
        style={{ height: 50, width: 100, marginBottom: 30 }}
      >
        Checkout
      </button>
    </article>
  )
}

export default RedirectToCheckout
```

## Server Endpoint Example

Your server endpoint should create the checkout session and return the `sessionId`:

```js
// pages/api/create-checkout-session.js (Next.js)
import Stripe from 'stripe'
const stripe = new Stripe(process.env.STRIPE_SECRET_KEY)

export default async function handler(req, res) {
  try {
    const { cartDetails } = req.body

    // Convert cart to line items
    const line_items = Object.values(cartDetails).map((item) => ({
      price_data: {
        currency: 'usd',
        product_data: {
          name: item.name,
          images: [item.image]
        },
        unit_amount: item.price
      },
      quantity: item.quantity
    }))

    // Create checkout session
    const session = await stripe.checkout.sessions.create({
      mode: 'payment',
      line_items,
      success_url: `${req.headers.origin}/success?session_id={CHECKOUT_SESSION_ID}`,
      cancel_url: req.headers.origin
    })

    // Return session ID to client
    res.json({ sessionId: session.id })
  } catch (error) {
    res.status(500).json({ error: error.message })
  }
}
```

## Notes

- Requires a server endpoint to create Stripe checkout sessions
- SessionId is required
- Initializes Stripe.js with the publishable key from CartProvider
- Returns error object if redirect fails
